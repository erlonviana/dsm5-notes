fun controleDeFluxo(notaMedia:Double){
    var status = if(notaMedia>=7.0)"Aprovado"else"Reprovado"
    //when é muito útil para valores específicos que vc já conhece
    when(notaMedia){
        10.0 ->{
            println("O aluno tirou 10")}
        //inserindo um range (intervalo)
        in 7.0..9.9->{println("O aluno passou com folga")}
        else ->{println("O aluno precisa estudar")}
    }
    println("O aluno está $status")}
